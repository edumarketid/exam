Skema SQL Database Supabase

Salin skema DDL berikut dan jalankan pada menu SQL Editor di dashboard Supabase Anda:

-- 1. TABEL DATA SISWA
CREATE TABLE IF NOT EXISTS siswa (
    id_siswa VARCHAR(50) PRIMARY KEY,
    nisn VARCHAR(20) UNIQUE NOT NULL,
    nama_siswa VARCHAR(150) NOT NULL,
    kelas VARCHAR(50) NOT NULL,
    status VARCHAR(20) DEFAULT 'Aktif',
    qr_code_key VARCHAR(50),
    ttl VARCHAR(100)
);

-- 2. TABEL AKUN LOGIN SISWA
CREATE TABLE IF NOT EXISTS akun_siswa (
    username VARCHAR(50) PRIMARY KEY,
    password VARCHAR(50) NOT NULL,
    id_siswa VARCHAR(50) REFERENCES siswa(id_siswa) ON DELETE CASCADE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 3. TABEL BANK SOAL (SUPPORTS HTML)
CREATE TABLE IF NOT EXISTS bank_soal (
    id BIGSERIAL PRIMARY KEY,
    mapel VARCHAR(100) NOT NULL,
    kelas VARCHAR(50) NOT NULL,
    soal_html TEXT NOT NULL,
    opsi_a TEXT NOT NULL,
    opsi_b TEXT NOT NULL,
    opsi_c TEXT NOT NULL,
    opsi_d TEXT NOT NULL,
    opsi_e TEXT NOT NULL,
    kunci_jawaban CHAR(1) NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 4. TABEL HASIL UJIAN SISWA
CREATE TABLE IF NOT EXISTS hasil_ujian (
    id BIGSERIAL PRIMARY KEY,
    id_siswa VARCHAR(50) REFERENCES siswa(id_siswa),
    mapel VARCHAR(100) NOT NULL,
    jawaban_json JSONB,
    nilai NUMERIC(5,2),
    status_selesai VARCHAR(30) DEFAULT 'SELESAI',
    jumlah_pelanggaran INT DEFAULT 0,
    waktu_selesai TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);
